/** * solura-track — email-open + document-open tracking worker * ───────────────────────────────────────────────────────── * Endpoints: * GET /px?d= → 1×1 gif, logs an email_open event * GET /doc?u=&d= → logs a doc_open event, 302 → target * GET /events?since=&token=… → JSON {events:[…]} newer than * * Setup (one time, ~5 minutes): * 1. dash.cloudflare.com → Workers & Pages → Create → Worker * Name it: solura-track → Deploy the hello-world → Edit code → * replace everything with this file → Deploy. * 2. Storage & Databases → KV → Create namespace: solura-track-events * 3. Worker → Settings → Bindings → Add → KV namespace: * Variable name TRACK_KV → select solura-track-events → Save. * 4. Worker → Settings → Variables & Secrets → Add: * Name TRACK_TOKEN → any long random string (this is your polling * password) → Save & Deploy. * 5. Copy the worker URL (https://solura-track..workers.dev) and * the token into the dashboard: ⚙ Settings → Data & analytics. * * Events auto-expire from KV after 30 days (the dashboard writes them * into the Comms Log sheet long before that). */ export default { async fetch(req, env) { const url = new URL(req.url); const cors = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, OPTIONS', 'Access-Control-Allow-Headers': '*' }; if (req.method === 'OPTIONS') return new Response(null, { headers: cors }); const readMeta = () => { try { return JSON.parse(atob(url.searchParams.get('d') || '')); } catch (_) { return {}; } }; const putEvent = async (type, extra) => { const ts = Date.now(); const id = ts + '-' + Math.random().toString(36).slice(2, 8); await env.TRACK_KV.put( 'ev:' + ts + ':' + id, JSON.stringify({ id, type, ts, ...readMeta(), ...extra }), { expirationTtl: 60 * 60 * 24 * 30 } ); return id; }; // ── Tracking pixel ── if (url.pathname === '/px') { await putEvent('email_open', { ua: req.headers.get('user-agent') || '' }); const gif = Uint8Array.from( atob('R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7'), c => c.charCodeAt(0) ); return new Response(gif, { headers: { 'content-type': 'image/gif', 'cache-control': 'no-store, max-age=0', ...cors } }); } // ── Tracked document redirect ── if (url.pathname === '/doc') { const target = url.searchParams.get('u') || ''; if (!/^https:\/\//.test(target)) return new Response('bad target', { status: 400, headers: cors }); await putEvent('doc_open', { target }); return Response.redirect(target, 302); } // ── Event feed for the dashboard ── if (url.pathname === '/events') { if (url.searchParams.get('token') !== env.TRACK_TOKEN) { return new Response(JSON.stringify({ error: 'unauthorized' }), { status: 401, headers: { 'content-type': 'application/json', ...cors } }); } const since = parseInt(url.searchParams.get('since') || '0', 10); const list = await env.TRACK_KV.list({ prefix: 'ev:', limit: 1000 }); const out = []; for (const k of list.keys) { const ts = parseInt(k.name.split(':')[1], 10); if (!(ts > since)) continue; const v = await env.TRACK_KV.get(k.name); if (v) { try { out.push(JSON.parse(v)); } catch (_) {} } } out.sort((a, b) => a.ts - b.ts); return new Response(JSON.stringify({ events: out }), { headers: { 'content-type': 'application/json', 'cache-control': 'no-store', ...cors } }); } return new Response('solura-track ok', { headers: cors }); } };
